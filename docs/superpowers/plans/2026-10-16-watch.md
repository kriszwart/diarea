# Diarea Watch v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `diarea watch <log>` reads a server access log and reports agent-style use of the API (parameter sweeps, id walks, sign-up bursts, route probes, machine cadence), scored like `scan`, without ever printing an IP, a query value or a raw user agent.

**Architecture:** `src/watch-parse.js` turns log lines (Combined Log Format or JSON lines) into privacy-safe events. `src/behaviour.js` is a pure rule engine: events in, findings, flagged actors and stats out. `src/watch.js` streams the file, applies `--since`, runs the engine and scores with `scoreOf` / `bandOf` from `src/checks.js`. `bin/diarea.js` gains a `watch` subcommand.

**Tech Stack:** Node 22+, plain ES modules, `node:crypto`, `node:net`, `node:readline`. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-16-watch-design.md`

## Global Constraints

- Node 22+, plain ES modules, no new dependencies.
- `npm test` runs without a browser or network.
- British English, no em dashes anywhere (code comments, copy, docs).
- Report tone cheeky but useful; the 💩 stays.
- Exit codes: 0 clean or under threshold, 1 when score >= `--fail-on`, 2 on tool error.
- Defensive only: Diarea reports; it never blocks, slows or answers traffic.
- No raw IP address, query value or raw user agent string in any report, text or JSON.
- Every finding has a stable `id` (never renamed), `sev` (`crit` | `warn` | `info`), `basis: "inferred"`, `title`, `detail`, `fix`, `evidence`.
- Error messages name paths as the user gave them, never an absolute local path.

## Review Focus

1. **Pagination and cache busters** (`?page=1..15`, `?_=<epoch ms>`) must never count as a parameter sweep. Pinned in Task 3.
2. **Personal data in paths** (`/users/jane@example.com/orders`) must be templated away before it can reach evidence. Pinned in Task 1.
3. **Big logs**: 100,000 events from one actor must analyse in under 5 seconds. Pinned in Task 4.
4. **Messy input**: blank lines, CRLF endings, a truncated JSON line and garbage are skipped and counted, never fatal. Pinned in Task 2.
5. **IPv6 clients**: actors hash, /48 networks group and IPv6 allowlist ranges match. Pinned in Task 2.

```js
/* The watcher's rule engine: parsed log events in, findings out. Pure: no
 * I/O, no clock, no randomness, so a middleware can reuse it later. Every
 * watcher rule lives here and only here; src/checks.js stays the engine for
 * frontend scans.
 *
 * An event is { t, actor, net, method, route, ids, params, status, ua }, made
 * by src/watch-parse.js. `params` holds raw query values for this run only;
 * nothing built here may copy a value into a finding.
 */

// Starting thresholds, tuned against the logs in test/watch.js.
export const LIMITS = {
  sweepValues: 12, sweepWindow: 10 * 60e3,
  enumIds: 25, enumWindow: 10 * 60e3, enumRun: 0.8,
  signups: 5, signupsAutomation: 10, signupWindow: 60 * 60e3,
  probeRoutes: 30, probeMissRate: 0.4,
  cadenceRequests: 200, cadenceAutomation: 100, cadenceCv: 0.15, cadenceWindow: 60 * 60e3,
};

export const BEHAVIOUR_BASIS = {
  "param-sweep": "inferred", "id-enumeration": "inferred", "signup-burst": "inferred",
  "route-probe": "inferred", "machine-cadence": "inferred",
};

export const BEHAVIOUR_SEV = {
  "param-sweep": "crit", "id-enumeration": "crit", "signup-burst": "warn",
  "route-probe": "warn", "machine-cadence": "info",
};

// The same four bands as scan, in the watcher's words.
export const WATCH_BANDS = {
  Sealed: "Nothing in this log reads like an agent at work.",
  Seeping: "A little automation about. Worth a look.",
  Leaking: "Someone is working your API out, one request at a time.",
  Gushing: "Something is reading your API like a manual.",
};

export const TIERS = ["watch", "slow", "challenge", "block"];

const ID_SEGMENT = /^(?:\d+|[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}|[0-9a-f]{16,}|(?=[^/]*\d)[A-Za-z0-9_-]{32,})$/i;

/** Template a path: ids, UUIDs, hashes, long segments and anything with an @ become :id. */
export function routeOf(pathname) {
  const ids = [];
  const segs = String(pathname || "").split("/").map((s) => {
    if (!s) return s;
    if (/^\d{1,15}$/.test(s)) { ids.push(Number(s)); return ":id"; }
    if (ID_SEGMENT.test(s) || s.length > 40 || /@|%40/i.test(s)) return ":id";
    return s;
  });
  return { route: segs.join("/") || "/", ids };
}

export const uniq = (a) => Array.from(new Set(a));

export function groupBy(items, keyOf) {
  const m = new Map();
  for (const x of items) {
    const k = keyOf(x);
    if (!m.has(k)) m.set(k, []);
    m.get(k).push(x);
  }
  return m;
}

export const minutes = (w) => `${Math.max(1, Math.round((w[w.length - 1].t - w[0].t) / 60e3))} min`;

// Windows no longer than `span` over items sorted by t, each starting halfway
// through the last, so a long log costs about two passes. Returns the first
// window with at least `min` items that `ok` accepts, or null.
export function findWindow(items, span, min, ok) {
  let j = 0;
  for (let i = 0; i < items.length; ) {
    if (j < i) j = i;
    while (j < items.length && items[j].t - items[i].t <= span) j++;
    const w = items.slice(i, j);
    if (w.length >= min && ok(w)) return w;
    if (j >= items.length) return null;
    i += Math.max(1, Math.floor((j - i) / 2));
  }
  return null;
}

function tierOf(ids) {
  const sevs = Array.from(ids, (id) => BEHAVIOUR_SEV[id]);
  if (sevs.includes("crit")) return "block";
  if (ids.has("signup-burst") || sevs.filter((s) => s === "warn").length >= 2) return "challenge";
  if (sevs.includes("warn")) return "slow";
  return "watch";
}

export function analyseEvents(events) {
  const ev = events.slice().sort((a, b) => a.t - b.t);
  const byActor = groupBy(ev, (e) => e.actor);
  const F = [];
  const flagged = new Map();
  // One finding per check; each hit is { actors: string[], line: string }.
  const add = (id, title, detail, fix, hits) => {
    if (!hits.length) return;
    F.push({ id, sev: BEHAVIOUR_SEV[id], basis: BEHAVIOUR_BASIS[id], title, detail, fix, evidence: hits.map((h) => h.line) });
    for (const h of hits) for (const a of h.actors) {
      if (!flagged.has(a)) flagged.set(a, new Set());
      flagged.get(a).add(id);
    }
  };

  // CHECKS

  const actors = Array.from(flagged, ([actor, ids]) => ({
    actor, ua: byActor.get(actor)[0].ua, events: byActor.get(actor).length, checks: Array.from(ids), tier: tierOf(ids),
  })).sort((a, b) => TIERS.indexOf(b.tier) - TIERS.indexOf(a.tier) || b.events - a.events);

  const iso = (t) => new Date(t).toISOString();
  return {
    findings: F,
    actors,
    stats: { events: ev.length, actors: byActor.size, from: ev.length ? iso(ev[0].t) : null, to: ev.length ? iso(ev[ev.length - 1].t) : null },
  };
}
```

Create `src/watch-parse.js`:

```js
/* Log lines to watcher events. Combined (or Common) Log Format and JSON
 * lines. The IP is hashed with a per-run salt and the user agent reduced to
 * a class before anything leaves this file; query values stay in memory.
 */
import crypto from "node:crypto";
import fs from "node:fs";
import path from "node:path";
import { isIP } from "node:net";
import { routeOf } from "./behaviour.js";
import { inV4, v6Groups } from "./guard.js";

const CLF = /^(\S+) \S+ \S+ \[([^\]]+)\] "([A-Za-z]+) (\S+)[^"]*" (\d{3}) \S+(?: "[^"]*" "([^"]*)")?/;
const MONTHS = { Jan: 0, Feb: 1, Mar: 2, Apr: 3, May: 4, Jun: 5, Jul: 6, Aug: 7, Sep: 8, Oct: 9, Nov: 10, Dec: 11 };

const FIELDS = {
  t: ["timestamp", "time", "ts", "EdgeStartTimestamp", "@timestamp"],
  ip: ["ip", "clientIp", "client_ip", "ClientIP", "remote_addr", "proxy.clientIp"],
  method: ["method", "ClientRequestMethod", "proxy.method", "request.method"],
  path: ["path", "url", "ClientRequestURI", "request.url", "proxy.path"],
  status: ["status", "statusCode", "EdgeResponseStatus", "proxy.statusCode"],
  ua: ["userAgent", "user_agent", "ClientRequestUserAgent", "proxy.userAgent"],
  auth: ["authorization", "headers.authorization", "request.headers.authorization", "proxy.authorization"],
};

const AUTOMATION = /curl|wget|python-requests|python-urllib|aiohttp|httpx|go-http-client|axios|node-fetch|undici|okhttp|java\/|libwww|headlesschrome|playwright|puppeteer|selenium|phantomjs|scrapy|postman/i;
const BOT = /bot\b|crawler|spider|slurp|facebookexternalhit/i;

export function uaClass(ua) {
  if (!ua || ua === "-") return "unknown";
  if (AUTOMATION.test(ua)) return "automation";
  if (BOT.test(ua)) return "bot";
  if (/mozilla\/5\.0/i.test(ua)) return "browser";
  return "unknown";
}

const pick = (o, keys) => {
  for (const k of keys) {
    const v = k.split(".").reduce((x, p) => (x == null ? undefined : x[p]), o);
    if (v !== undefined && v !== null && v !== "") return v;
  }
  return undefined;
};

// Seconds, milliseconds or nanoseconds since the epoch, or a date string.
function toTime(v) {
  const n = typeof v === "number" ? v : /^\d+(?:\.\d+)?$/.test(String(v)) ? Number(v) : NaN;
  if (Number.isFinite(n)) return n > 1e15 ? Math.round(n / 1e6) : n > 1e12 ? n : Math.round(n * 1000);
  return Date.parse(v);
}

function clfTime(s) {
  const m = /^(\d{2})\/(\w{3})\/(\d{4}):(\d{2}):(\d{2}):(\d{2}) ([+-])(\d{2})(\d{2})$/.exec(s);
  if (!m || !(m[2] in MONTHS)) return NaN;
  const utc = Date.UTC(+m[3], MONTHS[m[2]], +m[1], +m[4], +m[5], +m[6]);
  return utc - (m[7] === "+" ? 1 : -1) * (+m[8] * 60 + +m[9]) * 60e3;
}

function fromClf(line) {
  const m = CLF.exec(line);
  if (!m) return null;
  return { t: clfTime(m[2]), ip: m[1], method: m[3], path: m[4], status: Number(m[5]), ua: m[6] || "", auth: "" };
}

function fromJson(line) {
  let o;
  try { o = JSON.parse(line); } catch { return null; }
  if (!o || typeof o !== "object") return null;
  return {
    t: toTime(pick(o, FIELDS.t)), ip: String(pick(o, FIELDS.ip) ?? ""), method: String(pick(o, FIELDS.method) ?? ""),
    path: String(pick(o, FIELDS.path) ?? ""), status: Number(pick(o, FIELDS.status)), ua: String(pick(o, FIELDS.ua) ?? ""),
    auth: String(pick(o, FIELDS.auth) ?? ""),
  };
}

function splitPath(p) {
  let pathname = p, query = "";
  if (/^https?:\/\//i.test(p)) {
    try { const u = new URL(p); pathname = u.pathname; query = u.search.slice(1); } catch { return null; }
  } else {
    const q = p.indexOf("?");
    if (q >= 0) { pathname = p.slice(0, q); query = p.slice(q + 1); }
  }
  return { pathname, params: new Map(new URLSearchParams(query)) };
}

// The /24 (IPv4) or /48 (IPv6) an address sits in.
function networkOf(ip) {
  if (isIP(ip) === 4) return ip.split(".").slice(0, 3).join(".") + ".0/24";
  if (isIP(ip) === 6) return v6Groups(ip).slice(0, 3).map((g) => g.toString(16)).join(":") + "::/48";
  return ip;
}

function inCidr(ip, cidr) {
  const [base, b] = cidr.split("/");
  const bits = Number(b);
  if (isIP(ip) === 4 && isIP(base) === 4) return inV4(ip, base, bits);
  if (isIP(ip) !== 6 || isIP(base) !== 6) return false;
  const x = v6Groups(ip), y = v6Groups(base);
  for (let k = 0, left = bits; left > 0; k++, left -= 16) {
    const mask = left >= 16 ? 0xffff : (0xffff << (16 - left)) & 0xffff;
    if ((x[k] & mask) !== (y[k] & mask)) return false;
  }
  return true;
}

const validCidr = (c) => {
  const m = /^(.+)\/(\d{1,3})$/.exec(String(c));
  if (!m) return false;
  const v = isIP(m[1]);
  return (v === 4 && +m[2] <= 32) || (v === 6 && +m[2] <= 128);
};

/** Read an allowlist file. Errors name the file as given (basename) and the bad field. */
export function loadAllow(file) {
  const name = path.basename(file);
  let o;
  try { o = JSON.parse(fs.readFileSync(file, "utf8")); } catch { throw new Error(`Allowlist ${name} is not readable JSON.`); }
  const list = (k) => {
    const v = o[k] ?? [];
    if (!Array.isArray(v) || v.some((x) => typeof x !== "string" || !x)) throw new Error(`Allowlist ${name}: "${k}" must be a list of strings.`);
    return v;
  };
  const allow = { networks: list("networks"), userAgents: list("userAgents"), apiKeyPrefixes: list("apiKeyPrefixes") };
  const badNet = allow.networks.find((c) => !validCidr(c));
  if (badNet) throw new Error(`Allowlist ${name}: "networks" must hold CIDR ranges such as 203.0.113.0/24, not "${badNet}".`);
  return allow;
}

function isAllowed(r, allow) {
  if (allow.networks.some((c) => inCidr(r.ip, c))) return true;
  if (r.ua && allow.userAgents.some((s) => r.ua.includes(s))) return true;
  const key = r.auth.replace(/^(?:Bearer|Token|Basic)\s+/i, "");
  return Boolean(key) && allow.apiKeyPrefixes.some((s) => key.startsWith(s));
}

export function makeParser({ salt, allow = null }) {
  const hash = (s) => crypto.createHmac("sha256", salt).update(s).digest("hex").slice(0, 10);
  const p = { lines: 0, parsed: 0, allowed: 0, parseLine };
  function parseLine(line) {
    if (!line || !line.trim()) return null;
    p.lines++;
    const text = line.trim();
    const r = text.startsWith("{") ? fromJson(text) : fromClf(text);
    if (!r || !Number.isFinite(r.t) || !r.ip || !r.method || !r.path || !Number.isFinite(r.status)) return null;
    const split = splitPath(r.path);
    if (!split) return null;
    p.parsed++;
    if (allow && isAllowed(r, allow)) { p.allowed++; return null; }
    const { route, ids } = routeOf(split.pathname);
    return {
      t: r.t, actor: hash(r.ip + "\0" + r.ua), net: hash(networkOf(r.ip)), method: r.method.toUpperCase(),
      route, ids, params: split.params, status: r.status, ua: uaClass(r.ua),
    };
  }
  return p;
}
```

```js
// Paging, cursors and cache busters step through values on purpose.
const PAGING = /^(?:page|p|pg|offset|skip|start|limit|per_?page|size|page_?size|cursor|after|before|from|_|t|ts|timestamp|v|cb)$/i;
const NUMERIC = /^-?\d+(?:\.\d+)?$/;

const dedupe = (vals) => vals.filter((v, k) => k === 0 || v !== vals[k - 1]);
const gaps = (vals) => vals.slice(1).map((v, k) => Math.abs(v - vals[k])).filter((d) => d > 0);
const median = (a) => { const s = a.slice().sort((x, y) => x - y); return s[Math.floor(s.length / 2)]; };

function evenSteps(vals) {
  const d = gaps(vals);
  if (d.length < 4) return false;
  const m = median(d);
  return d.filter((x) => Math.abs(x - m) <= m * 0.1).length / d.length >= 0.75;
}

function halving(vals) {
  const d = gaps(vals);
  if (d.length < 4) return false;
  let n = 0;
  for (let k = 1; k < d.length; k++) { const r = d[k] / d[k - 1]; if (r >= 0.35 && r <= 0.65) n++; }
  return n / (d.length - 1) >= 0.6;
}

function runRatio(ids) {
  const v = uniq(ids).sort((a, b) => a - b);
  let inRuns = 0, len = 1;
  for (let k = 1; k <= v.length; k++) {
    if (k < v.length && v[k] - v[k - 1] === 1) len++;
    else { if (len >= 3) inRuns += len; len = 1; }
  }
  return v.length ? inRuns / v.length : 0;
}

function push(map, key, item) {
  if (!map.has(key)) map.set(key, []);
  map.get(key).push(item);
}

function sweeps(ev) {
  const groups = new Map();
  for (const e of ev) for (const [name, raw] of e.params) {
    if (PAGING.test(name) || !NUMERIC.test(raw)) continue;
    const v = Number(raw);
    if (Math.abs(v) >= 1e9) continue; // epoch times, not a probe
    push(groups, `${e.actor}\0${e.method} ${e.route}\0${name}`, { t: e.t, v });
  }
  const hits = [];
  for (const [key, items] of groups) {
    let kind = "";
    const w = findWindow(items, LIMITS.sweepWindow, LIMITS.sweepValues, (w) => {
      if (new Set(w.map((x) => x.v)).size < LIMITS.sweepValues) return false;
      const vals = dedupe(w.map((x) => x.v));
      kind = evenSteps(vals) ? "even steps" : halving(vals) ? "halving the gap" : "";
      return Boolean(kind);
    });
    if (!w) continue;
    const [actor, where, name] = key.split("\0");
    hits.push({ actors: [actor], line: `actor ${actor}: ${where} ?${name}, ${new Set(w.map((x) => x.v)).size} values in ${minutes(w)}, ${kind}` });
  }
  return hits;
}

function walks(ev) {
  const groups = new Map();
  for (const e of ev) {
    if (e.ua === "bot") continue;
    e.ids.forEach((v, k) => push(groups, `${e.actor}\0${e.method} ${e.route}\0${k}`, { t: e.t, v }));
  }
  const hits = [];
  for (const [key, items] of groups) {
    const w = findWindow(items, LIMITS.enumWindow, LIMITS.enumIds, (w) => {
      const ids = w.map((x) => x.v);
      return new Set(ids).size >= LIMITS.enumIds && runRatio(ids) >= LIMITS.enumRun;
    });
    if (!w) continue;
    const [actor, where] = key.split("\0");
    const ids = w.map((x) => x.v);
    hits.push({ actors: [actor], line: `actor ${actor}: ${where}, ${new Set(ids).size} ids in ${minutes(w)}, ${Math.round(runRatio(ids) * 100)}% consecutive` });
  }
  return hits;
}
```

```js
function bursts(ev) {

// Running sums over an hour-long window, so this stays one pass per actor.
function cadence(byActor) {
  const hits = [];
  for (const [actor, list] of byActor) {
    if (list[0].ua === "bot") continue;
    const ts = list.map((e) => e.t);
    const automation = list[0].ua === "automation";
    let i = 0, sum = 0, sq = 0, found = null;
    for (let j = 1; j < ts.length && !found; j++) {
      const g = ts[j] - ts[j - 1];
      sum += g; sq += g * g;
      while (ts[j] - ts[i] > LIMITS.cadenceWindow) { const h = ts[i + 1] - ts[i]; sum -= h; sq -= h * h; i++; }
      const n = j - i;
      const count = n + 1;
      if (automation && count >= LIMITS.cadenceAutomation) found = `${count} requests in an hour from an automation client`;
      else if (count >= LIMITS.cadenceRequests && sum > 0) {
        const mean = sum / n;
        const cv = Math.sqrt(Math.max(0, sq / n - mean * mean)) / mean;
        if (cv < LIMITS.cadenceCv) found = `${count} requests in an hour, ${Math.round(mean / 1000)}s apart, near-identical gaps`;
      }
    }
    if (found) hits.push({ actors: [actor], line: `actor ${actor}: ${found}` });
  }
  return hits;
}
```

```js
/* diarea watch: read a server log, find agent-style use of the API, score it
 * like a scan. Reads only; never touches traffic.
 */
import crypto from "node:crypto";
import fs from "node:fs";
import readline from "node:readline";
import { makeParser } from "./watch-parse.js";
import { analyseEvents, WATCH_BANDS } from "./behaviour.js";
import { scoreOf, bandOf } from "./checks.js";

const UNITS = { m: 60e3, h: 3600e3, d: 86400e3 };

/** "30m", "24h", "7d" to milliseconds, or null. */
export function parseDuration(s) {
  const m = /^(\d+)([mhd])$/.exec(String(s ?? ""));
  return m && Number(m[1]) > 0 ? Number(m[1]) * UNITS[m[2]] : null;
}

export async function watchLines(lines, { since = null, allow = null, salt = crypto.randomBytes(16).toString("hex") } = {}) {
  const parser = makeParser({ salt, allow });
  let events = [];
  for await (const line of lines) {
    const e = parser.parseLine(line);
    if (e) events.push(e);
  }
  if (!parser.lines) throw new Error("The log is empty.");
  if (!parser.parsed) throw new Error(`No line could be read as Combined Log Format or JSON lines (${parser.lines} tried).`);
  if (since != null && events.length) {
    // Relative to the newest event, so an old log replays the same way every time.
    const newest = events.reduce((m, e) => (e.t > m ? e.t : m), -Infinity);
    events = events.filter((e) => e.t >= newest - since);
  }
  const { findings, actors, stats } = analyseEvents(events);
  const score = scoreOf(findings);
  const band = bandOf(score);
  return {
    tool: "diarea", mode: "watch", score,
    band: { label: band.label, line: WATCH_BANDS[band.label] },
    stats: { ...stats, lines: parser.lines, skipped: parser.lines - parser.parsed, allowlisted: parser.allowed },
    findings, actors,
  };
}

/** Watch a log file, or stdin when target is "-". Errors name the target as given. */
export async function watch(target, opts = {}) {
  let input = process.stdin;
  if (target !== "-") {
    let st;
    try { st = fs.statSync(target); } catch { throw new Error(`Cannot read ${target}.`); }
    if (!st.isFile()) throw new Error(`${target} is not a file. Give a log file, or - for stdin.`);
    input = fs.createReadStream(target, "utf8");
  }
  const rl = readline.createInterface({ input, crlfDelay: Infinity });
  return watchLines(rl, opts);
}
```

```js
function renderWatch(r, s) {
  const findings = r.findings.slice().sort((a, b) => SEV_ORDER[a.sev] - SEV_ORDER[b.sev]);
  const bandColour = r.band.label === "Sealed" ? s.good : r.band.label === "Seeping" ? s.warn : s.crit;
  const st = r.stats;
  const when = (iso) => (iso ? iso.slice(0, 16).replace("T", " ") : "");
  const out = [""];
  out.push(`  ${s.mag("\u{1F4A9} diarea watch")}  ${s.dim(`${st.events} requests from ${st.actors} actors${st.from ? `, ${when(st.from)} to ${when(st.to)} UTC` : ""}`)}`);
  out.push(`  ${s.bold(`Watch Score ${r.score}/100`)}  ${bandColour(r.band.label)}  ${s.dim(r.band.line)}`);
  if (st.skipped || st.allowlisted) out.push(`  ${s.dim(`${st.skipped} lines skipped, ${st.allowlisted} allowlisted`)}`);
  out.push("");
  if (!findings.length) out.push(`  ${s.good("Nothing in this log reads like an agent at work.")}`, "");
  for (const f of findings) out.push(...findingLines(f, s));
  if (r.actors.length) {
    out.push(`  ${s.bold("Who, and what to do about them")}`);
    for (const a of r.actors.slice(0, 10)) {
      const tier = a.tier === "block" ? s.crit(a.tier.padEnd(9)) : a.tier === "watch" ? s.info(a.tier.padEnd(9)) : s.warn(a.tier.padEnd(9));
      out.push(`  ${tier} actor ${a.actor}  ${s.dim(`${a.ua}, ${a.events} requests: ${a.checks.join(", ")}`)}`);
    }
    if (r.actors.length > 10) out.push(`  ${s.dim(`...and ${r.actors.length - 10} more (--json for all)`)}`);
    out.push("");
  }
  out.push(`  ${s.dim("Tiers are advice. Diarea reads logs; it never touches your traffic. Actor ids change every run.")}`);
  out.push("");
  return out.join("\n");
}

async function runWatch(o) {
  if (o.sarif) o.errors.push("--sarif is not available for watch yet. Use --json.");
  if (o.storageState) o.errors.push("--storage-state only applies to scan.");
  if (o.errors.length) { console.error(o.errors.join("\n") + "\n" + HELP); return 2; }
  const target = o._[1];
  if (!target) { console.error("Nothing to watch. Give a log file, or - for stdin.\n" + HELP); return 2; }
  const { watch, parseDuration } = await import("../src/watch.js");

  let since = null;
  if (o.since != null) {
    since = parseDuration(o.since);
    if (since == null) { console.error(`--since must look like 30m, 24h or 7d, not "${o.since}".`); return 2; }
  }
  let report;
  try {
    report = await watch(target, { since, allow: o.allow ? loadAllow(o.allow) : null });

  if (o.failOn != null && report.score >= o.failOn) return 1;
  if (o.strict && report.findings.some((f) => f.sev === "crit")) return 1;
```
